6. Jaccard Similarity Analysis.
CODE:
def jaccard_matrix(X):
    n = X.shape[0]
    J = np.zeros((n , n))
    for i in range(n):
        for j in range(n):
            J[i][j] = jaccard_score(X[i], X[j]) 
    return J

jac_sim = jaccard_matrix(binary_matrix.values)     #Jaccard Similarity= (A intersection B) / (A U B)
jac_df = pd.DataFrame(jac_sim, index = incidence.index, columns = incidence.index)
jac_df
